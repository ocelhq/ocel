## Zero infrastructure code

You write minimal config naming the platform to deploy to via a provider, never infrastructure code for deploying your app. Swapping providers is easy:

<Compare>
<div>

```ts tab="TypeScript" tab-group="config" title="ocel.config.ts" lineNumbers
import { defineConfig } from "ocel/config";
import awsProvider from "ocel/providers/aws"; // [!code --]

export default defineConfig({
  slug: "acme",
  provider: awsProvider(), // [!code --]
  apps: [{ name: "web", path: "." }],
});
```

```json tab="JSON" title="ocel.json" lineNumbers
{
  "$schema": "https://ocel.dev/schema/ocel.schema.json",
  "slug": "acme",
  "provider": "aws", // [!code --]
  "apps": [{ "name": "web", "path": "." }]
}
```

</div>
<div>

```ts tab="TypeScript" tab-group="config" title="ocel.config.ts" lineNumbers

  slug: "acme",

  apps: [{ name: "web", path: "." }],
```


## Application defined infrastructure

A resource (database, bucket) is a function call in your app. Declare it once in a [discovery folder](/docs/configuration#discoverypaths), import it where you use it.

<Tabs groupId="lang" persist items={["TypeScript", "Go", "Python", "Rust"]}>

<Tab value="TypeScript">

```ts title="resources/index.ts"
import { postgres } from "ocel/postgres";

export const db = postgres("main");
```

```ts title="src/server.ts"
import express from "express";
import { db } from "../resources";

const app = express();

app.get("/documents", async (_req, res) => {
  // Use it directly in your app code! 
  const { rows } = await db.query("select * from documents");
  res.json(rows);
});

app.listen(3000);
```

</Tab>

<Tab value="Go">

```go title="resources/resources.go"
package resources

import "ocel.dev"

var DB = ocel.Postgres("main")
```

```go title="main.go"
package main

import (
	"net/http"

	"acme.dev/web/resources"
)

func documents(w http.ResponseWriter, r *http.Request) {
	pool, err := resources.DB.Pool(r.Context())
	if err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}
	rows, err := pool.Query(r.Context(), "select * from documents")
	// ...
}
```

</Tab>

<Tab value="Python">

```python title="resources/__init__.py"
import ocel

db = ocel.postgres("main")
```

```python title="main.py"
from fastapi import FastAPI
from resources import db

app = FastAPI()


@app.get("/documents")
async def documents():
    rows = await db.fetch("select * from documents")
    return [dict(row) for row in rows]
```

</Tab>

<Tab value="Rust">

```rust title="src/main.rs"
#[derive(ocel::Resources)]
struct Infra {
    main: ocel::Postgres,
}

#[ocel::main]
#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    let resources = Infra::load()?;
    let rows = sqlx::query("select * from documents")
        .fetch_all(resources.main.pool().await?)
        .await?;
    // ...
    Ok(())
}
```

</Tab>

</Tabs>
